\section{\textbf{Research Interests}}
\vspace{1mm}
\small{
I work on AI safety, with a focus on \textbf{reward hacking} in large language models: a model gets a high reward in ways people did not intend, without actually solving the task. My work spans four questions:
\begin{itemize}
\item \textbf{Science of reward hacking:} How reward hacking emerges in RL and generalizes into broader misalignment.
\item \textbf{Alignment training:} Training models that stay aligned under imperfect rewards or specifications.
\item \textbf{Monitoring systems:} Detecting misbehavior at runtime via chain-of-thought monitors and activation probes.
\item \textbf{Evaluation and its science:} Measuring these reliably when models can recognize evaluation.
\end{itemize}
}

\item[\textbf{[2]}] Can Jin*, \underline{\textbf{Rui Wu*}}, Tong Che*, Qixin Zhang, Hongwu Peng, Jiahui Zhao, Zhenting Wang, Wenqi Wei, Ligong Han, Zhao Zhang, Yuan Cao, Ruixiang Tang, Dimitris N. (2026). \textbf{Reasoning over Precedents Alongside Statutes: Case-Augmented Deliberative Alignment for LLM Safety}. \textit{ACL 2026 Main} [\href{https://arxiv.org/abs/2601.08000}{arXiv}] \\
{\footnotesize\color{subtle}TL;DR: Finds that detailed safety codes give inconsistent safety gains and reduce helpfulness in open models, and proposes RL on safety reasoning with simple rules plus illustrative cases.}

\item[\textbf{[3]}] \underline{\textbf{Rui Wu}}, Yihao Quan, Zeru Shi, Zhenting Wang, Yanshu Li, Ruixiang Tang. (2025). \textbf{Read the Scene, Not the Script: Outcome-Aware Safety for LLMs}. \textit{NeurIPS 2025 ResponsibleFM Workshop} [\href{https://arxiv.org/abs/2510.04320}{arXiv}] \\
{\footnotesize\color{subtle}TL;DR: Identifies consequence-blindness, where models judge safety by surface cues rather than actual outcomes; builds a benchmark to measure it and a dataset that reduces both jailbreaks and over-refusal.}

\item[\textbf{[4]}] Can Jin, Jiakang Li, \underline{\textbf{Rui Wu}}, Eddy Zhang, Dimitris N Metaxas. (2026). \textbf{Weak Critics Make Strong Learners: On-Policy Critique Distillation for Scalable Oversight}. \textit{ICML 2026 AI4Math Workshop} [\href{https://arxiv.org/abs/2606.00424}{arXiv}] \\
{\footnotesize\color{subtle}TL;DR: Uses a weak model as a critic that suggests revision directions rather than answers, and distills critic-guided behavior into the strong model through on-policy training.}

\end{enumerate}
